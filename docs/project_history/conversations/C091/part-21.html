function Get-ProjectManifest {
    param(
        [Parameter(Mandatory)]
        [string]$Root
    )

    @(
        Get-ChildItem `
            -LiteralPath $Root `
            -File `
            -Recurse `
            -Force |
        ForEach-Object {
            [pscustomobject]@{
                relative_path =
                    $_.FullName.Substring($Root.Length + 1).Replace('\','/')

                bytes = $_.Length

                sha256 = (
                    Get-FileHash `
                        -LiteralPath $_.FullName `
                        -Algorithm SHA256
                ).Hash.ToLowerInvariant()
            }
        } |
        Sort-Object {
            $_.relative_path.ToLowerInvariant()
        }
    )
}
