$rels = @(Import-Csv -LiteralPath $input)

if ($rels.Count -ne 32) {
    throw "STOP: expected 32 physical relationships; found $($rels.Count)."

function QTable([string]$name) {
    "'" + $name.Replace("'","''") + "'"
}

function QColumn([string]$table,[string]$column) {
    "$(QTable $table)[$column]"
}

$parts = @()

foreach ($r in $rels) {

    $fromColumn = QColumn $r.FromTable $r.FromColumn
    $toColumn   = QColumn $r.ToTable   $r.ToColumn

    $parts += @"
ROW(
    "Relationship", "$($r.Relationship)",
    "FromTable", "$($r.FromTable)",
    "FromColumn", "$($r.FromColumn)",
    "ToTable", "$($r.ToTable)",
    "ToColumn", "$($r.ToColumn)",
    "IsActive", "$($r.IsActive)",
    "NonBlankFromKeys",
        COUNTROWS(
            FILTER(
                VALUES($fromColumn),
                NOT ISBLANK($fromColumn)
            )
        ),
    "OrphanKeyCount",
        COUNTROWS(
            FILTER(
                VALUES($fromColumn),
                NOT ISBLANK($fromColumn)
                    &&
                NOT CONTAINS(
                    ALL($toColumn),
                    $toColumn,
                    $fromColumn
                )
            )
        )
)
"@
}

$query = @"
EVALUATE
VAR __Orphans =
    UNION(
$($parts -join ",`r`n")
    )
RETURN
    ADDCOLUMNS(
        __Orphans,
        "Pass", [OrphanKeyCount] = 0
    )
ORDER BY [Relationship]
"@

[IO.File]::WriteAllText(
    $output,
    $query,
    [Text.UTF8Encoding]::new($false)
)
